    65→  @Post('signup')
    66→  async signup(@Body() signupDto: SignupDto, @Request() req: any, @Res({ passthrough: true }) res: Response) {
    67→    // Securite: l'endpoint public d'auto-inscription est limite aux roles acteurs.
    68→    // Les roles a privileges (back-office, admin, super_admin) ne sont jamais creables
    69→    // sans authentification: ils passent par les endpoints administres (ex: POST /users/admin).
    70→    if (!ACTEUR_ROLES.includes(signupDto.role)) {
    71→      throw new ForbiddenException("Ce role ne peut pas etre cree via l'inscription publique");
    72→    }
    73→    // Politique mot de passe canonique JULABA: BO = 123456, acteur = 0000, mustChangePassword=true.
    74→    // Le password recu du frontend est ignore et ecrase pour garantir la regle uniforme.
    75→    signupDto.password = getDefaultPasswordForRole(signupDto.role);
    76→    (signupDto as any).mustChangePassword = true;
    77→    const deviceInfo = req.headers['user-agent'];
    78→    const ipAddress = req.ip || req.headers['x-forwarded-for'];
    79→    const result = await this.authService.signup(signupDto, deviceInfo, ipAddress);
    80→    this.setTokenCookies(res, result.accessToken, result.refreshToken);
    81→    // Jetons aussi dans le corps (auth mobile sans cookie cross-domaine).
    82→    return { user: { ...result.user, mustChangePassword: result.user.mustChangePassword ?? false }, success: true, accessToken: result.accessToken, refreshToken: result.refreshToken };
    83→  }

   130→  @Post('refresh')
   131→  @HttpCode(HttpStatus.OK)
   132→  async refresh(@Request() req: any, @Body() body: any, @Res({ passthrough: true }) res: Response) {
   133→    const token = req.cookies?.refresh_token || body?.refreshToken;
   134→    if (!token) return { error: 'Refresh token manquant' };
   135→    try {
   136→      const deviceInfo = req.headers['user-agent'];
   137→      const ipAddress = req.ip || req.headers['x-forwarded-for'];
   138→      const result = await this.authService.rotateRefreshToken(token, deviceInfo, ipAddress);
   139→        const isBO = BO_ROLES.includes(result.user?.role);
   140→      this.setTokenCookies(res, result.accessToken, result.refreshToken, isBO);
   141→      // LE JETON DE RAFRAÎCHISSEMENT PART AUSSI DANS LE CORPS — HYGIÈNE-1 axe 2.
   142→      // `login` le fait déjà, et pour une raison écrite juste au-dessus : sur
   143→      // mobile, les cookies cross-domaine (julaba-web ↔ julaba-api) sont
   144→      // bloqués. `refresh` ne l'avait jamais suivi. Conséquence dans l'APK : le
   145→      // téléphone rejouait indéfiniment le MÊME jeton stocké, puisqu'il n'en
   146→      // recevait jamais le suivant. Or la rotation marque l'ancien « used », et
   147→      // rejouer un jeton « used » est traité comme une COMPROMISSION :
   148→      // `rotateRefreshToken` révoque alors TOUTES les sessions de la marchande.
   149→      // Une session qui se renouvelle normalement finissait donc par déconnecter
   150→      // partout celle qui vend. La rotation est maintenant complète des deux
   151→      // côtés : celui qui présente un jeton reçoit son successeur.
   152→      return { success: true, accessToken: result.accessToken, refreshToken: result.refreshToken };
   153→    } catch (e) {
   154→      res.clearCookie('access_token', this.getTokenCookieBaseOptions());
   155→      res.clearCookie('refresh_token', this.getTokenCookieBaseOptions());
   156→      res.clearCookie('bo_access_token', this.getTokenCookieBaseOptions());
   157→      return { error: e.message || 'Token invalide' };
   158→    }
   159→  }

   171→  @SkipThrottle()
   172→  @Get('sessions')
   173→  @UseGuards(JwtAuthGuard)
   174→  async getSessions(@Request() req: any) {
   175→    const sessions = await this.tokenRepo.find({
   176→      where: { userId: req.user.id, revoked: false, used: false },
   177→      order: { createdAt: 'DESC' },
   178→      take: 10,
   179→    });
   180→    const crypto = require('crypto');
   181→    const salt = process.env.REFRESH_TOKEN_SALT;
   182→    const rawToken = req.cookies?.refresh_token || '';
   183→    const currentHash = rawToken
   184→      ? (salt ? crypto.createHmac('sha256', salt).update(rawToken).digest('hex') : '')
   185→      : '';
   186→    return {
   187→      sessions: sessions.map(s => ({
   188→        id: s.id,
   189→        deviceInfo: s.deviceInfo || 'Appareil inconnu',
   190→        ipAddress: s.ipAddress || 'IP inconnue',
   191→        createdAt: s.createdAt,
   192→        expiresAt: s.expiresAt,
   193→        isCurrent: currentHash ? s.tokenHash === currentHash : false,
   194→      })),
   195→    };
   196→  }

   231→  async setPin(@Request() req: any, @Body() body: { pin: string; currentPin?: string }) {
   232→    if (!body.pin || !/^\d{4}$/.test(body.pin)) {
   233→      return { success: false, message: 'Le code PIN doit contenir exactement 4 chiffres' };
   234→    }
   235→    const bcrypt = require('bcryptjs');
   236→    const user = await this.userRepo.findOne({ where: { id: req.user.id } });
   237→    if (!user) return { success: false, message: 'Utilisateur introuvable' };
   238→    if (user.pinSecurityEnabled && user.pinCodeHash) {
   239→      if (!body.currentPin) return { success: false, message: 'Code PIN actuel requis' };
   240→      const valid = await bcrypt.compare(body.currentPin, user.pinCodeHash);
   241→      if (!valid) return { success: false, message: 'Code PIN actuel incorrect' };
   242→    }
   243→    const hash = await bcrypt.hash(body.pin, 10);
   244→    await this.userRepo.update(req.user.id, {
   245→      pinCodeHash: hash,
   246→      pinSecurityEnabled: true,
   247→    } as any);
   248→    return { success: true };

   251→  @Post('pin/verify')
   252→  @UseGuards(JwtAuthGuard)
   253→  @HttpCode(HttpStatus.OK)
   254→  async verifyPin(@Request() req: any, @Body() body: { pin: string }) {
   255→    if (!body.pin) return { valid: false };
   256→    const bcrypt = require('bcryptjs');
   257→    // Recharger le user complet (failedPinAttempts / lockedUntil à jour).
   258→    const user = await this.userRepo.findOne({ where: { id: req.user.id } });
   259→    if (!user || !user.pinCodeHash) return { valid: false };
   260→
   261→    // Verrou hérité de l'ancienne politique (100 ans) : la règle qui l'a posé
   262→    // n'existe plus, on le lève au lieu de le subir.
   263→    if (estVerrouHeriteSansFin(user.lockedUntil)) {
   264→      await this.userRepo.update(user.id, { failedPinAttempts: 0, lockedUntil: null });
   265→      user.lockedUntil = null;
   266→      user.failedPinAttempts = 0;
   267→    }
   268→
   269→    // Attente en cours : on ne vérifie même pas le PIN, et on dit combien il reste.
   270→    if (user.lockedUntil && user.lockedUntil.getTime() > Date.now()) {
   271→      return { valid: false, locked: true, attenteMs: user.lockedUntil.getTime() - Date.now() };
   272→    }
   273→
   274→    const valid = await bcrypt.compare(body.pin, user.pinCodeHash);
   275→    if (valid) {
   276→      await this.userRepo.update(user.id, { failedPinAttempts: 0, lockedUntil: null });
   277→      return { valid: true };
   278→    }
   279→
   280→    const attempts = (user.failedPinAttempts ?? 0) + 1;
   281→    const attente = attenteApresEchecs(attempts);
   282→    if (attente > 0) {
   283→      await this.userRepo.update(user.id, {
   284→        failedPinAttempts: attempts,
   285→        lockedUntil: new Date(Date.now() + attente),
   286→      });
   287→      return { valid: false, locked: true, attenteMs: attente };
   288→    }
   289→    await this.userRepo.update(user.id, { failedPinAttempts: attempts });
   290→    return { valid: false, locked: false, essaisRestants: essaisAvantAttente(attempts) };
   291→  }

   504→  @Post('identificateur/me/verify-pin')
   505→  @UseGuards(JwtAuthGuard, RolesGuard)
   506→  @Roles('identificateur')
   507→  @HttpCode(HttpStatus.OK)
   508→  async verifyIdentificateurPin(@Body() body: { pin: string }, @Request() req: any) {
   509→    if (!body.pin || !/^\d{4}$/.test(body.pin)) {
   510→      return { valid: false, message: 'Le PIN doit contenir exactement 4 chiffres' };
   511→    }
   512→    const user = await this.userRepo.findOne({ where: { id: req.user.id } });
   513→    if (!user) return { valid: false, message: 'Utilisateur introuvable' };
   514→    if ((user as any).role !== 'identificateur') {
   515→      return { valid: false, message: 'Cet utilisateur n\'est pas un identificateur' };
   516→    }
   517→    if (!(user as any).pinCodeEncryptedIdentificateur) {
   518→      return { valid: false, message: 'Aucun PIN défini pour cet identificateur' };
   519→    }
   520→
   521→    // ── SEC-07 : CETTE ROUTE N'AVAIT AUCUN COMPTEUR ──────────────────────────
   522→    //
   523→    // Elle comparait en temps constant — bien — puis acceptait un nombre
   524→    // ILLIMITÉ d'essais sur 4 096 combinaisons. Quelques minutes suffisaient à
   525→    // les épuiser depuis une session ouverte, par exemple sur un téléphone
   526→    // volé non verrouillé. Le PIN identificateur protège la modification des
   527→    // fiches acteurs : ce n'était pas une protection, c'était un péage.
   528→    //
   529→    // L'arbitrage « 4 chiffres, alphabet 2–9 » (cf. pin-identificateur.ts) est
   530→    // posé SUR ce verrou. Sans lui, il ne tient pas. Ce sont donc deux moitiés
   531→    // du même choix, et les tests vérifient les deux ensemble.
   532→    const verrou = await this.attenteVerrouPinIdentificateur(user);
   533→    if (verrou) return { valid: false, ...verrou };
   534→
   535→    try {
   536→      const pin = this.pinCrypto.decrypt((user as any).pinCodeEncryptedIdentificateur);
   537→      const pinBuffer = Buffer.from(pin.trim().padEnd(4, '\0'));
   538→      const inputBuffer = Buffer.from(body.pin.trim().padEnd(4, '\0'));
   539→      const valid = pinBuffer.length === inputBuffer.length &&
   540→        timingSafeEqual(pinBuffer, inputBuffer);
   541→      if (valid) {
   542→        await this.reussitePinIdentificateur(user.id);
   543→        return { valid: true };
   544→      }
   545→      return { valid: false, ...(await this.echecPinIdentificateur(user)) };
   546→    } catch {
   547→      return { valid: false, message: 'Erreur lors de la vérification du PIN' };
   548→    }
   549→  }
   550→
   551→  // ── LE VERROU DU PIN IDENTIFICATEUR, EN UN SEUL ENDROIT ───────────────────
   552→  //
   553→  // Il utilise l'échelle de `verrou-pin.ts` (3 échecs → 5 min, 6 → 15 min,
   554→  // 9 et au-delà → 1 h, jamais définitif) mais SES PROPRES COLONNES.
   555→  //
   556→  // Pourquoi pas `failedPinAttempts` / `lockedUntil` ? Parce qu'une connexion
   557→  // par mot de passe réussie les remet à zéro (`auth.service.login`). Partager
   558→  // ces champs offrirait le contournement exact que Patrick a demandé de
   559→  // rendre impossible : rater trois fois le PIN, se déconnecter, se
   560→  // reconnecter, recommencer indéfiniment. Ici, seul un PIN JUSTE remet le
   561→  // compteur à zéro — ou une réinitialisation, qui change le secret.
   562→
   563→  private async attenteVerrouPinIdentificateur(
   564→    user: User,
   565→  ): Promise<{ locked: true; attenteMs: number } | null> {
   566→    const jusqua = (user as any).identificateurPinLockedUntil as Date | null;
   567→    if (estVerrouHeriteSansFin(jusqua)) {
   568→      await this.userRepo.update(user.id, {
   569→        failedIdentificateurPinAttempts: 0,
   570→        identificateurPinLockedUntil: null,
   571→      } as any);
   572→      (user as any).identificateurPinLockedUntil = null;
   573→      (user as any).failedIdentificateurPinAttempts = 0;
   574→      return null;
   575→    }
   576→    if (jusqua && jusqua.getTime() > Date.now()) {
   577→      return { locked: true, attenteMs: jusqua.getTime() - Date.now() };
   578→    }
   579→    return null;
   580→  }

   582→  private async reussitePinIdentificateur(id: string): Promise<void> {
   583→    await this.userRepo.update(id, {
   584→      failedIdentificateurPinAttempts: 0,
   585→      identificateurPinLockedUntil: null,
   586→    } as any);
   587→  }

   589→  private async echecPinIdentificateur(
   590→    user: User,
   591→  ): Promise<{ locked: boolean; attenteMs?: number; essaisRestants?: number }> {
   592→    const essais = ((user as any).failedIdentificateurPinAttempts ?? 0) + 1;
   593→    const attente = attenteApresEchecs(essais);
   594→    if (attente > 0) {
   595→      await this.userRepo.update(user.id, {
   596→        failedIdentificateurPinAttempts: essais,
   597→        identificateurPinLockedUntil: new Date(Date.now() + attente),

   599→      return { locked: true, attenteMs: attente };
   600→    }
   601→    await this.userRepo.update(user.id, { failedIdentificateurPinAttempts: essais } as any);
   602→    return { locked: false, essaisRestants: essaisAvantAttente(essais) };
   603→  }

   700→    }

   888→        const newCredential = {
   889→          credentialID: credential.id,
   890→          credentialPublicKey: Buffer.from(credential.publicKey).toString('base64url'),
   891→          counter: credential.counter,
   892→          deviceType: credentialDeviceType,
   893→          backedUp: credentialBackedUp,
   894→          transports: body.response.transports || [],

   896→        const existing = user.webauthnCredentials || [];
   897→        await this.userRepo.update(req.user.id, {
   898→          webauthnCredentials: [...existing, newCredential],
   899→          webauthnChallenge: null,
   900→        } as any);
   901→        return { verified: true };
   902→      }
   903→      return { verified: false };
   904→    } catch (e: any) {

   911→  async webauthnAuthOptions(@Body() body: { phone: string }) {
   912→    if (!body.phone) return { error: 'Numéro requis' };
   913→    const trimmed = body.phone.trim();
   914→    const digits = trimmed.replace(/\D/g, '');
   915→    const phone = trimmed.startsWith('+225')
   916→      ? trimmed.replace(/\s/g, '')
   917→      : digits.startsWith('225')
   918→        ? `+${digits}`
   919→        : digits.startsWith('0')
   920→          ? `+225${digits}`
   921→          : `+225${digits}`;
   922→    const user = await this.userRepo.findOne({ where: { phone } });
   923→    if (!user) return { error: 'Utilisateur introuvable' };
   924→    if (!user.webauthnCredentials?.length) {
   925→      return { error: 'Aucune clé biométrique enregistrée pour ce compte' };
   926→    }
   927→    const allowCredentials = (user.webauthnCredentials || []).map((c) => ({
   928→      id: c.credentialID,
   929→      transports: c.transports as any,
   930→    }));
   931→    const options = await generateAuthenticationOptions({
   932→      rpID: process.env.WEBAUTHN_RP_ID || 'julaba.online',
   933→      userVerification: 'preferred',
   934→      allowCredentials: allowCredentials as any,
   935→    });
   936→    await this.userRepo.update(user.id, { webauthnChallenge: options.challenge } as any);
   937→    return { ...options, userId: user.id };
   938→  }

   940→  @Post('webauthn/authenticate/verify')
   941→  @HttpCode(HttpStatus.OK)
   942→  async webauthnAuthVerify(
   943→    @Body() body: { response: AuthenticationResponseJSON; userId: string },
   944→    @Res({ passthrough: true }) res: Response,
   945→    @Request() req: any,
   946→  ) {
   947→    if (!body.userId || !body.response) return { verified: false, error: 'Données manquantes' };
   948→    const user = await this.userRepo.findOne({ where: { id: body.userId } });
   949→    if (!user || !user.webauthnChallenge) return { verified: false, error: 'Challenge manquant' };
   950→    const credentialID = body.response.id;
   951→    const credential = (user.webauthnCredentials || []).find((c) => c.credentialID === credentialID);
   952→    if (!credential) return { verified: false, error: 'Credential introuvable' };
   953→    try {
   954→      const { verified, authenticationInfo } = await verifyAuthenticationResponse({
   955→        response: body.response,
   956→        expectedChallenge: user.webauthnChallenge,
   957→        expectedOrigin: process.env.WEBAUTHN_ORIGIN || 'https://julaba.online',
   958→        expectedRPID: process.env.WEBAUTHN_RP_ID || 'julaba.online',
   959→        credential: {
   960→          id: credential.credentialID,
   961→          publicKey: Buffer.from(credential.credentialPublicKey, 'base64url'),
   962→          counter: credential.counter,
   963→          transports: credential.transports as any,
   964→        },
   965→      });
   966→      if (verified) {
   967→        const updated = (user.webauthnCredentials || []).map((c) =>
   968→          c.credentialID === credentialID
   969→            ? { ...c, counter: authenticationInfo.newCounter }
   970→            : c,
   971→        );
   972→        await this.userRepo.update(user.id, {
   973→          webauthnCredentials: updated,
   974→          webauthnChallenge: null,
   975→        } as any);
   976→        const ipAddress = req.ip || req.headers['x-forwarded-for'];
   977→        const result = await this.authService.loginById(user.id, req.headers['user-agent'], ipAddress as string);
   978→        const isBO = BO_ROLES.includes(result.user?.role);
   979→        this.setTokenCookies(res, result.accessToken, result.refreshToken, isBO);
   980→        // Jetons AUSSI dans le corps (comme /login) : sur mobile les cookies
   981→        // cross-domaine sont bloqués -> sans ça, l'empreinte « réussit » puis la
   982→        // requête suivante est 401 et l'utilisateur retombe sur l'écran de login.
   983→        return {
   984→          verified: true,
   985→          user: result.user,
   986→          accessToken: result.accessToken,
   987→          refreshToken: result.refreshToken,
   988→        };
   989→      }
   990→      return { verified: false };
   991→    } catch (e: any) {
   992→      return { verified: false, error: e.message };
   993→    }
   994→  }

   996→  private getTokenCookieBaseOptions() {
   997→    const isProd = process.env.NODE_ENV === 'production';
   998→    // Multi-domaines (frontend et backend sur des domaines DIFFÉRENTS, ex. Render V2) :
   999→    // le navigateur n'envoie le cookie cross-site que si SameSite=None + Secure.
  1000→    // En PRODUCTION on prend donc 'none' par défaut (le déploiement V2 est cross-domaine
  1001→    // et sert en HTTPS) — sans quoi le cookie de session n'est jamais renvoyé et
  1002→    // l'utilisateur paraît « jamais connecté ». En dev : 'lax' (même origine).
  1003→    // Toujours surchargeable via COOKIE_SAMESITE.
  1004→    const sameSite =
  1005→      (process.env.COOKIE_SAMESITE as 'lax' | 'none' | 'strict') ||
  1006→      (isProd ? 'none' : 'lax');
  1007→    const secure = isProd || sameSite === 'none';
  1008→    return { httpOnly: true, secure, sameSite, path: '/' };
  1009→  }

  1011→  private setTokenCookies(res: Response, accessToken: string, refreshToken: string, isBO = false) {
  1012→    const base = this.getTokenCookieBaseOptions();
  1013→    const tokenName = isBO ? 'bo_access_token' : 'access_token';
  1014→    res.cookie(tokenName, accessToken, { ...base, maxAge: 24 * 60 * 60 * 1000 });
  1015→    res.cookie('refresh_token', refreshToken, { ...base, maxAge: 7 * 24 * 60 * 60 * 1000 });
  1016→  }
